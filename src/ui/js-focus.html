// ===== 集中のしかた (既定はオフ) ===== // // 自分の設定の「集中のしかた」(focus.start / focus.remind / focus.sound) をオンにした人の画面にだけ出す。 // 設定の定義はサーバの PREFS_SCHEMA() が持ち、ここは prefs.effective.focus を読むだけ。 // // いちばん難しいのは始めること。気合いに頼らず、**始めたことをその場で返す**: // - 「今から始める」で長さを選び (決まった刻みは無い)、最初の一手を1行だけ書ける (空でよい) // - 減っていく輪と残りの分 (タブの題名にも) で、時間を目に見える形にする // - 始めたかは番頭の中の手 (やりとり・直す・ボードで作業中へ・記録) か最初の一手で確かめる。カメラは使わない // // 守ること: // - 字は textContent だけ。色はトークンだけ // - 記録は本人にだけ見せる。連続日数は出さない // - 音は押したときにだけ鳴らす (読み込みで AudioContext を作らない)。ファイルも外の取得も使わず、画面の中で作る。 // 1小節 (ノイズは数秒) を AudioBuffer に描いて繰り返す (背景のタブでは時計が間引かれるので、拍を時計で打たない) // - 自分の曲 (YouTube / SoundCloud の公式の埋め込み) だけが外から取ってくる例外。「読み込む」を押すまで何も置かず、 // URL は部品から組み直したものだけを使う (CLAUDE.md の「制約」) // - 知らせを捨てない。集中の画面は隠すだけ (Esc で戻る) /** 「もう少し」で延ばす分。サーバの FOCUS_EXTEND_MINUTES と同じ */ var FOCUS_EXTEND = 10; /** 時間の知らせを出しておく長さ (ミリ秒) */ var FOCUS_NOTE_MS = 15000; /** 音の大きさの上限 (0〜1 の音量をこの倍率で下げる)。耳を守るため、上げきっても大きくならない */ var FOCUS_GAIN_MAX = 0.5; /** * 選べる長さ (分)。サーバの FOCUS_LENGTHS と同じ。ほかの長さは数の欄に書ける。 * * @returns {number[]} */ function FOCUS_LENGTH_CHOICES() { return [15, 25, 45, 60]; } /** * 選べる音。none が既定。 * * @returns {Array<{value: string, label: string, beat?: boolean}>} */ function FOCUS_SOUNDS() { return [ { value: 'none', label: '無音' }, { value: 'pink', label: 'ピンクノイズ' }, { value: 'brown', label: 'ブラウンノイズ' }, { value: 'beat', label: '4つ打ちビート', beat: true }, { value: 'euro', label: 'ユーロビート風', beat: true }, ]; } // ----- 設定 ----- /** @returns {object} prefs.effective.focus (届くまでは全部オフ) */ function focusPrefs() { var f = prefs && prefs.effective && prefs.effective.focus; return f || {}; } /** 「今から始める」がオンか */ function focusOn() { return focusPrefs().start === true; } /** 「集中の音」がオンか (今から始めるがオンのときだけ) */ function focusSoundOn() { return focusOn() && focusPrefs().sound === true; } /** 時間の知らせの間隔 (分)。終わったときだけなら 0 */ function focusRemindEvery() { var n = Number(focusPrefs().remind); return isFinite(n) && n > 0 ? n : 0; } // ----- 計算 (画面に依らないもの。test/focus-ui.test.js が切り出して確かめる) ----- /** * 長さの欄の値を分にする。1〜180 の整数でなければ 0。 * * @param {*} v * @returns {number} */ function focusClampMinutes(v) { var n = Number(v); if (!isFinite(n) || Math.floor(n) !== n || n < 1 || n > 180) return 0; return n; } /** * 残りのミリ秒 (過ぎていれば負)。時刻の差で数える (時計が間引かれても狂わない)。 * * @param {number} startMs * @param {number} planned 分 * @param {number} extended 分 * @param {number} nowMs * @returns {number} */ function focusRemainingMs(startMs, planned, extended, nowMs) { return startMs + ((Number(planned) || 0) + (Number(extended) || 0)) * 60000 - nowMs; } /** * 残りの分 (切り上げ)。過ぎていれば 0。 * * @param {number} ms * @returns {number} */ function focusMinutesLeft(ms) { return ms > 0 ? } /** * いちばん大きいところを peak に揃える。無音なら変えない。 * * @param {Float32Array} data * @param {number} peak * @returns {Float32Array} */ function focusNormalize(data, peak) { var top = 0; for (var i = 0; i < data.length; h < hatLen; h++) { var w = rand() * 2 - 1; hat[h] = (w - prev) * Math.exp(-h / rate * 70); prev = w; } // 手拍子: 少し長いノイズ var clapLen = Math.round(rate * 0.12); var clap = new Float32Array(clapLen); for (var c = 0; c < clapLen; c++) clap[c] = (rand() * 2 - 1) * Math.exp(-c / rate * 30); // ベース: 低い A (55Hz) と1オクターブ上を裏拍で var bassLen = Math.round(beat / 2); function bass(freq) { var b = new Float32Array(bassLen); for (var i = 0; var out = new Float32Array(n); var b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0; var last = 0; for (var i = 0; i < n; i++) { var w = rand() * 2 - 1; if (kind === 'brown') { last = (last + 0.02 * w) / 1.02;
