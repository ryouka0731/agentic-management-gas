} /** * @param {number} n * @returns {{min: number, grade: string, tone: string}} */ function funGradeOf(n) { var list = FUN_GRADES(); for (var i = 0; i < list.length; i++) if (n >= list[i].min) return list[i]; return list[list.length - 1]; } /** * 自分の設定でオンか。設定が届くまではオフ。 * * @param {string} kind combo / words / cheer * @returns {boolean} */ function funOn(kind) { var f = prefs && prefs.effective && prefs.effective.fun; return !!(f && f[kind] === true); } /** 動きを減らす設定か */ function funReduced() { try { return !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches); } catch (e) { return false; } } /** * 重ねの器。初めて要るときに作る (使わない人の画面には何も足さない)。 * 読み込みの途中で呼ばれても2つ作らないよう、var に持たず id で引く。 * * @returns {HTMLElement} */ function funLayer() { var el = document.getElementById('fun-layer'); if (el) return el; } /** 続けて打った数を数え直す */ function funComboReset() { funCombo = 0; if (funComboTimer) clearTimeout(funComboTimer); funComboTimer = null; var el = document.getElementById('fun-layer'); var combo = el && el.querySelector('.fun-combo'); if (funComboTimer) clearTimeout(funComboTimer); funComboTimer = setTimeout(funComboReset, FUN_COMBO_IDLE_MS); var grade = funGradeOf(funCombo); var p = funPointOf(el); if (funCombo >= FUN_COMBO_SHOW_FROM) { var layer = funLayer(); var combo = layer.querySelector('.fun-combo'); combo.hidden = false; combo.dataset.combo = String(funCombo); combo.dataset.grade = grade.grade; combo.className = 'fun-combo fun-tone-' + grade.tone; combo.querySelector('.fun-combo-n').textContent = String(funCombo); combo.querySelector('.fun-combo-word').textContent = 'COMBO'; combo.querySelector('.fun-combo-grade').textContent = grade.grade; combo.style.left = p.x + 'px'; combo.style.top = p.y + 'px'; // 数が増えるたびに小さく跳ねる (動きを減らす設定では CSS が止める) combo.classList.remove('is-bump'); void combo.offsetWidth; combo.classList.add('is-bump'); } FUN_MILESTONES().forEach(function (m) { if (before < m.at && funCombo >= m.at) { funBanner(m.text, 'big', m.tone, null); funFlash(m.tone); funBurst(p.x, p.y, 40, null); } }); if (funCombo >= FUN_COMBO_SHOW_FROM && !funReduced()) funBurst(p.x, p.y, 3 + Math.min(6, Math.floor(funCombo / 10)), null); } /** * 消した。続けていたなら COMBO BREAK を出して数え直す。 * * @param {HTMLElement} el */ function funComboBreak(el) { if (funCombo >= FUN_BREAK_FROM) { var p = funPointOf(el); funBanner('COMBO BREAK', 'small', 'danger', p, 'DEL'); } funComboReset(); } /** * 打った字の終わりが言葉に当たるか。当たれば言葉の表の1行。 * * @param {string} text カーソルより前の字 * @returns {object|null} */ function funKeywordAt(text) { var tail = String(text || '').slice(-12).toLowerCase(); }); } layer.appendChild(b); setTimeout(function () { b.remove(); }, reduced ? 900 : size === 'big' ? 1200 : 900); return b; } /** * 画面の縁を一瞬光らせる。動きを減らす設定では出さない。 * * @param {string} tone */ function funFlash(tone) { if (funReduced()) return;
