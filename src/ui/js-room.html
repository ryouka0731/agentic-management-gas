// ===== 作業部屋 (既定はオフ) ===== // // 自分の設定の「作業部屋を使う」(room.on) をオンにした人の画面にだけ出す。設定の定義はサーバの PREFS_SCHEMA() が持つ。 // 雨・キーボード・たき火・風・カフェのざわめきを流し、推しが机で作業している部屋を描く。 // // 守ること: // - 音は**画面の中で作る** (ファイルも外の取得も使わない)。押した聞き手の中でだけ作って鳴らす (自動では鳴らさない)。 // でたらめに起きる音 (雨粒・打鍵・はぜる音) も、20〜30秒を AudioBuffer に描いて繰り返す // (背景のタブでは時計が間引かれるので、時計で鳴らさない)。音の形は画面に依らない関数にしてあり、 // test/room-ui.test.js が切り出して確かめる // - 大きさは ROOM_GAIN_MAX を超えない。始めと止めはゆっくり上げ下げする // - 集中の音・自分の曲とは同時に流さない (片方を始めたら、もう片方を止める) // - 字は textContent だけ。色はトークンだけ。動きを減らす設定では、雨も揺れも止めた絵にする // - 部屋は集中を始めない。集中しているあいだだけ、隅に残りの輪を出す /** 全体の大きさの上限 (0〜1 の大きさをこの倍率で下げる)。耳を守るため、上げきっても大きくならない */ var ROOM_GAIN_MAX = 0.5; /** 音を描く1秒の標本の数。まわりの音には十分で、描く手間が半分になる */ var ROOM_RATE = 24000; /** 始めるとき・止めるときに上げ下げする秒 */ var ROOM_FADE_IN = 2; var ROOM_FADE_OUT = 1.5; /** 窓の雨粒の数の上限 */ var ROOM_DROPS_MAX = 60; /** 推しが手を止めて別の顔をする長さ (ミリ秒) */ var ROOM_GLANCE_MS = 4000; /** * 流せる音。並びは画面の並び。seconds は描いて繰り返す長さ。 * * @returns {Array<{key: string, label: string, seconds: number}>} */ function ROOM_SOUNDS() { return [ { key: 'rain', label: '雨', seconds: 24 }, { key: 'keys', label: 'キーボード', seconds: 24 }, { key: 'fire', label: 'たき火', seconds: 20 }, { key: 'wind', label: '風', seconds: 30 }, { key: 'cafe', label: 'カフェのざわめき', seconds: 30 }, ]; } // ----- 設定 ----- /** 作業部屋がオンか。設定が届くまではオフ */ function roomOn() { var r = prefs && prefs.effective && prefs.effective.room; return !!(r && r.on === true); } // ----- 計算 (画面に依らないもの。test/room-ui.test.js が切り出して確かめる) ----- /** * 0〜1 に収める。数でなければ def。 * * @param {*} v * @param {number} def * @returns {number} */ function roomClamp01(v, def) { if (typeof v !== 'number' && typeof v !== 'string') return def; var n = Number(v); if (v === '' || !isFinite(n)) return def; return Math.max(0, Math.min(1, n)); } /** 全体の大きさ (0〜1) を、実際に掛ける倍率にする。上限を超えない */ function roomMasterGain(v) { return roomClamp01(v, 0) * ROOM_GAIN_MAX; } /** * 既定の混ぜ方。最初は雨だけ。 * * @returns {{master: number, sounds: Object}} */ function roomMixDefault() { var vol = { rain: 0.6, keys: 0.45, fire: 0.5, wind: 0.4, cafe: 0.4 }; var sounds = {}; ROOM_SOUNDS().forEach(function (s) { sounds[s.key] = { on: s.key === 'rain', volume: vol[s.key] }; }); return { master: 0.6, sounds: sounds }; } /** * 端末に覚えた混ぜ方を読める形にそろえる。知らない音は捨て、欠けた音は既定で埋める。 * * @param {*} raw * @returns {{master: number, sounds: Object}} */ function roomMixClean(raw) { var def = roomMixDefault(); if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return def; var given = raw.sounds && typeof raw.sounds === 'object' ? raw.sounds : {}; var out = { master: roomClamp01(raw.master, def.master), sounds: {} }; ROOM_SOUNDS().forEach(function (s) { var d = def.sounds[s.key]; var r = Object.prototype.hasOwnProperty.call(given, s.key) ? given[s.key] : null; out.sounds[s.key] = r && typeof r === 'object' ? { on: r.on === true, volume: roomClamp01(r.volume, d.volume) } : { on: d.on, volume: d.volume }; }); return out; } /** * 時刻から窓の外の様子。朝 (5〜9時)・昼 (10〜15時)・夕方 (16〜18時)・夜。 * * @param {number} hour 0〜23 (端末の時計) * @returns {string} morning / day / evening / night */ function roomPhaseOf(hour) { var h = Number(hour); if (h >= 5 && h < 10) return 'morning'; if (h >= 10 && h < 16) return 'day'; if (h >= 16 && h < 19) return 'evening'; return 'night'; } /** @returns {string} 窓の外の様子の名前 */ function roomPhaseLabel(phase) { return { morning: '朝', day: '昼', evening: '夕方', night: '夜' }[phase] || ''; } /** * ノイズを count 個。white / pink / brown。 * * @param {string} kind * @param {number} count * @param {function(): number} rand * @returns {Float32Array} */ function roomNoise(kind, count, rand) { var out = new Float32Array(count); var b0 = 0, b1 = 0, b2 = 0, last = 0; for (var i = 0; i < count; i++) { var w = rand() * 2 - 1; if (kind === 'brown') { last = (last + 0.02 * w) / 1.02; out[i] = last * 3.5; } else if (kind === 'pink') { b0 = 0.99765 * b0 + w * 0.0990460; b1 = 0.96300 * b1 + w * 0.2965164; b2 = 0.57000 * b2 + w * 1.0526913; out[i] = (b0 + b1 + b2 + w * 0.1848) * 0.2; } else { out[i] = w; } } return out; } /** * 2次のフィルタの係数 (lowpass / highpass / bandpass)。 * * @param {string} type * @param {number} freq Hz * @param {number} q * @param {number} rate * @returns {{b0: number, b1: number, b2: number, a1: number, a2: number}} */ function roomBiquad(type, freq, q, rate) { var f = Math.max(10, Math.min(Number(freq) || 10, rate * 0.45)); var w0 = 2 * Math.PI * f / rate; var cos = Math.cos(w0); var alpha = Math.sin(w0) / (2 * Math.max(0.1, Number(q) || 0.7)); var b0, b1, b2; if (type === 'lowpass') { b0 = (1 - cos) / 2; b1 = 1 - cos; b2 = (1 - cos) / 2; } else if (type === 'highpass') { b0 = (1 + cos) / 2; b1 = -(1 + cos); b2 = (1 + cos) / 2; } else { b0 = alpha; b1 = 0; b2 = -alpha; } var a0 = 1 + alpha; return { b0: b0 / a0, b1: b1 / a0, b2: b2 / a0, a1: -2 * cos / a0, a2: (1 - alpha) / a0 }; } /** * フィルタを通す (その場で書き換える)。 * * @param {Float32Array} data * @param {{b0: number, b1: number, b2: number, a1: number, a2: number}} c * @returns {Float32Array} */ function roomFilter(data, c) { var x1 = 0, x2 = 0, y1 = 0, y2 = 0; for (var i = 0; i < data.length; i++) { var x = data[i]; var y = c.b0 * x + c.b1 * x1 + c.b2 * x2 - c.a1 * y1 - c.a2 * y2; x2 = x1; x1 = x; y2 = y1; y1 = y; data[i] = y; } return data; } /** * ゆっくり揺れる値の節。roomSmoothAt で読む。一周すると頭に戻るので、繰り返しても継ぎ目が出ない。 * * @param {number} steps 一周の節の数 * @param {function(): number} rand * @returns {number[]} */ function roomSmoothPoints(steps, rand) { var n = Math.max(2, Math.round(steps)); var out = []; for (var i = 0; i < n; i++) out.push(rand()); return out; } /** * 節の間をなめらかにつないだ値 (0〜1)。x は一周を 1 とした位置。 * * @param {number[]} points * @param {number} x * @returns {number} */ function roomSmoothAt(points, x) { var n = points.length; var pos = (x - Math.floor(x)) * n; var k = Math.floor(pos); var t = pos - k; var a = points[k % n]; var b = points[(k + 1) % n]; return a + (b - a) * (1 - Math.cos(Math.PI * t)) / 2; } /** * 短い音を重ねる。はみ出したぶんは頭へ回り込ませる (繰り返すので継ぎ目で切れない)。 * * @param {Float32Array} out * @param {number} at * @param {Float32Array} sample * @param {number} gain */ function roomPut(out, at, sample, gain) { var n = out.length; var start = ((Math.floor(at) % n) + n) % n; for (var i = 0; i < sample.length; i++) out[(start + i) % n] += sample[i] * gain; } /** * 短く減っていく音 (音程のある「ぽつ」)。noise の割合だけざらつかせる。 * * @param {number} rate * @param {number} freq * @param {number} decay 秒 * @param {function(): number} rand * @param {number} noise 0〜1 * @returns {Float32Array} */ function roomPing(rate, freq, decay, rand, noise) { var len = Math.max(1, Math.round(rate * decay * 6)); var out = new Float32Array(len); var phase = rand() * 2 * Math.PI; for (var i = 0; i < len; i++) { var t = i / rate; var env = Math.min(1, t / 0.0005) * Math.exp(-t / decay); out[i] = (Math.sin(phase + 2 * Math.PI * freq * t) * (1 - noise) + (rand() * 2 - 1) * noise) * env; } return out; } /** * 帯を絞ったノイズの短いはじけ (「かち」)。 * * @param {number} rate * @param {number} freq * @param {number} q * @param {number} decay 秒 * @param {function(): number} rand * @returns {Float32Array} */ function roomClick(rate, freq, q, decay, rand) { var len = Math.max(2, Math.round(rate * (decay * 6 + 0.002))); var out = new Float32Array(len); for (var i = 0; i < len; i++) out[i] = (rand() * 2 - 1) * Math.exp(-i / rate / decay); return focusNormalize(roomFilter(out, roomBiquad('bandpass', freq, q, rate)), 1); } /** * 雨。高いところを抑えたノイズの上に、雨粒の「ぽつ」と、窓の縁から落ちる重いしずくを重ねる。 * * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array} */ function roomRenderRain(rate, seconds, rand) { var n = Math.round(rate * seconds); var f = Math.round(rate * 0.1); var bed = roomNoise('pink', n + f, rand); roomFilter(bed, roomBiquad('lowpass', 2800, 0.7, rate)); roomFilter(bed, roomBiquad('highpass', 180, 0.7, rate)); var out = focusNormalize(focusLoopSeam(bed, f), 0.5); var swell = roomSmoothPoints(seconds / 3, rand); for (var i = 0; i < n; i++) out[i] *= 0.7 + 0.3 * roomSmoothAt(swell, i / n); var drops = Math.round(seconds * 45); for (var d = 0; d < drops; d++) { var at = Math.floor(rand() * n); var amp = 0.04 + 0.3 * Math.pow(rand(), 3); var s = rand() < 0.7 ? roomPing(rate, 2200 + 3500 * rand(), 0.002 + 0.004 * rand(), rand, 0.6) : roomClick(rate, 3000 + 3000 * rand(), 2, 0.0015, rand); roomPut(out, at, s, amp); } var drips = Math.round(seconds * 1.2); for (var k = 0; k < drips; k++) { roomPut(out, Math.floor(rand() * n), roomPing(rate, 700 + 600 * rand(), 0.012 + 0.01 * rand(), rand, 0.2), 0.25 + 0.2 * rand()); } return focusNormalize(out, 0.8); } /** * キーを1つ打つ音 (押した「かち」と底の「とん」、少しあとに離した軽い「かち」)。space はスペースキー (低く大きい)。 * * @param {Float32Array} out * @param {number} at * @param {number} rate * @param {function(): number} rand * @param {boolean} space */ function roomKeyStroke(out, at, rate, rand, space) { var amp = space ? 1 : 0.55 + 0.35 * rand(); roomPut(out, at, roomClick(rate, space ? 1300 : 2200 + 1600 * rand(), 1.4, 0.0018, rand), amp); roomPut(out, at, roomPing(rate, space ? 120 : 170 + 80 * rand(), 0.007, rand, 0.15), amp * 0.45); var up = at + Math.round(rate * (0.05 + 0.04 * rand())); roomPut(out, up, roomClick(rate, 3200 + 1500 * rand(), 1.8, 0.0012, rand), amp * 0.35); } /** * キーボード。ひとしきり打って (1.5〜6秒)、手を止める (0.7〜3.3秒) をくり返す。言葉の切れ目にスペースキー。 * * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array} */ function roomRenderKeys(rate, seconds, rand) { var n = Math.round(rate * seconds); var f = Math.round(rate * 0.1); var tone = roomNoise('brown', n + f, rand); roomFilter(tone, roomBiquad('lowpass', 300, 0.7, rate)); var out = focusNormalize(focusLoopSeam(tone, f), 0.03); var t = 0.2 + rand() * 0.6; while (t < seconds - 0.4) { var end = Math.min(seconds - 0.4, t + 1.5 + rand() * 4.5); while (t < end) { var space = rand() < 0.14; roomKeyStroke(out, Math.floor(t * rate), rate, rand, space); t += space ? 0.18 + rand() * 0.2 : 0.08 + rand() * 0.13; } t += 0.7 + rand() * 2.6; } return focusNormalize(out, 0.8); } /** * たき火。低いごうごうという音 (揺れる) と細かいしゅうしゅう、ぱちっとはぜる音 (ときどき続けて)、まれに大きくはぜる音。 * * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array} */ function roomRenderFire(rate, seconds, rand) { var n = Math.round(rate * seconds); var f = Math.round(rate * 0.1); var bed = roomNoise('brown', n + f, rand); roomFilter(bed, roomBiquad('lowpass', 500, 0.7, rate)); var out = focusNormalize(focusLoopSeam(bed, f), 0.45); var hiss = roomNoise('white', n + f, rand); roomFilter(hiss, roomBiquad('highpass', 2500, 0.7, rate)); hiss = focusNormalize(focusLoopSeam(hiss, f), 0.05); var roar = roomSmoothPoints(seconds * 1.5, rand); for (var i = 0; i < n; i++) { var x = i / n; out[i] = out[i] * (0.55 + 0.45 * roomSmoothAt(roar, x)) + hiss[i] * roomSmoothAt(roar, x + 0.37); } var pops = Math.round(seconds * 7); for (var p = 0; p < pops; p++) { var at = Math.floor(rand() * n); var cluster = rand() < 0.3 ? 2 + Math.floor(rand() * 3) : 1; for (var c = 0; c < cluster; c++) { roomPut(out, at + Math.round(c * rate * (0.008 + 0.02 * rand())), roomClick(rate, 900 + 3200 * rand(), 0.9, 0.0006 + 0.0015 * rand(), rand), 0.15 + 0.75 * Math.pow(rand(), 2)); } } var cracks = Math.round(seconds * 0.35); for (var k = 0; k < cracks; k++) { roomPut(out, Math.floor(rand() * n), roomClick(rate, 500 + 700 * rand(), 0.7, 0.006 + 0.006 * rand(), rand), 0.8 + 0.3 * rand()); } return focusNormalize(out, 0.8); } /** * 風。ノイズの帯の高さをゆっくり動かし (すき間を抜ける音)、強弱を波のように揺らす。細い笛のような音を少し重ねる。 * * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array} */ function roomRenderWind(rate, seconds, rand) { var n = Math.round(rate * seconds); var f = Math.round(rate * 0.1); var src = roomNoise('white', n + f, rand); var raw = new Float32Array(n + f); var tone = roomSmoothPoints(seconds / 4, rand); var whistle = roomSmoothPoints(seconds / 5, rand); var gust = roomSmoothPoints(seconds / 2.5, rand); var c = null, c2 = null, wv = 0; var x1 = 0, x2 = 0, y1 = 0, y2 = 0, u1 = 0, u2 = 0, v1 = 0, v2 = 0; for (var i = 0; i < n + f; i++) { if (i % 64 === 0) { // 一周の中の位置で決める (繰り返したとき、同じ位置は同じ音色になる) var x = (i % n) / n; c = roomBiquad('bandpass', 260 + 640 * roomSmoothAt(tone, x), 0.9, rate); c2 = roomBiquad('bandpass', 900 + 500 * roomSmoothAt(whistle, x), 9, rate); wv = 0.35 * roomSmoothAt(whistle, x + 0.5); } var w = src[i]; var y = c.b0 * w + c.b1 * x1 + c.b2 * x2 - c.a1 * y1 - c.a2 * y2; x2 = x1; x1 = w; y2 = y1; y1 = y; var v = c2.b0 * w + c2.b1 * u1 + c2.b2 * u2 - c2.a1 * v1 - c2.a2 * v2; u2 = u1; u1 = w; v2 = v1; v1 = v; raw[i] = y + v * wv; } var out = focusLoopSeam(raw, f); for (var j = 0; j < n; j++) out[j] *= 0.25 + 0.75 * Math.pow(roomSmoothAt(gust, j / n), 1.5); return focusNormalize(out, 0.8); } /** * カフェのざわめき。帯を絞ったノイズの「声らしいもの」を6つ、音節ほどの速さで強弱を付け、話す・黙るをくり返す。 * 言葉は作らない (聞き取れる声は無い)。ときどき器の触れ合う「ちん」。 * * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array} */ function roomRenderCafe(rate, seconds, rand) { var n = Math.round(rate * seconds); var f = Math.round(rate * 0.1); var bed = roomNoise('pink', n + f, rand); roomFilter(bed, roomBiquad('lowpass', 700, 0.7, rate)); var out = focusNormalize(focusLoopSeam(bed, f), 0.12); for (var v = 0; v < 6; v++) { var hi = roomNoise('white', n + f, rand); var lo = new Float32Array(hi); roomFilter(lo, roomBiquad('bandpass', 280 + 420 * rand(), 1.4, rate)); roomFilter(hi, roomBiquad('bandpass', 1100 + 1300 * rand(), 2.2, rate)); for (var a = 0; a < n + f; a++) lo[a] += hi[a] * 0.35; var voice = focusNormalize(focusLoopSeam(lo, f), 1); var syl = roomSmoothPoints(seconds * (3.5 + 2 * rand()), rand); var phrase = roomSmoothPoints(seconds / (2 + 3 * rand()), rand); var level = 0.25 + 0.35 * rand(); for (var i = 0; i < n; i++) { var x = i / n; var s = roomSmoothAt(syl, x); var p = roomSmoothAt(phrase, x); var gate = p < 0.4 ? 0 : p > 0.6 ? 1 : (p - 0.4) / 0.2; out[i] += voice[i] * s * s * gate * level; } } var clinks = Math.round(seconds * 0.3); for (var k = 0; k < clinks; k++) { var at = Math.floor(rand() * n); var freq = 2600 + 1600 * rand(); roomPut(out, at, roomPing(rate, freq, 0.09 + 0.06 * rand(), rand, 0.02), 0.12 + 0.08 * rand()); roomPut(out, at, roomPing(rate, freq * 2.76, 0.05, rand, 0), 0.05); } return focusNormalize(out, 0.8); } /** * 音を描く。 * * @param {string} key ROOM_SOUNDS の key * @param {number} rate * @param {number} seconds * @param {function(): number} rand * @returns {Float32Array|null} 知らない key なら null */ function roomRender(key, rate, seconds, rand) { var fn = { rain: roomRenderRain, keys: roomRenderKeys, fire: roomRenderFire, wind: roomRenderWind, cafe: roomRenderCafe }[key]; return fn ? fn(rate, seconds, rand) : null; } // ----- 混ぜ方を覚える (この端末にだけ) ----- var roomMixNow = null; /** @returns {object} 覚えた混ぜ方 (無ければ既定) */ function roomMixRead() { try { return roomMixClean(JSON.parse(window.localStorage.getItem('roomMix') || 'null')); } catch (e) { return roomMixDefault(); } } /** @param {object} mix */ function roomMixSave(mix) { try { window.localStorage.setItem('roomMix', JSON.stringify(mix)); } catch (e) { /* 覚えられなくても流せる */ } } /** @returns {object} いまの混ぜ方 */ function roomMix() { if (!roomMixNow) roomMixNow = roomMixRead(); return roomMixNow; } // ----- 音 (押したときにだけ作る) ----- /** {ctx, master, voices: {key: {src, gain}}, playing} */ var roomAudio = null; /** 描いた音 (key → AudioBuffer)。同じ音の仕組みのあいだは描き直さない */ var roomBuffers = {}; /** 鳴っているか */ function roomSoundPlaying() { return !!(roomAudio && roomAudio.playing); } /** * 鳴らす。**押した聞き手の中から呼ぶ。** 集中の音と自分の曲は止める。 * * @returns {boolean} 鳴らし始めたか */ function roomSoundPlay() { var Ctor = focusAudioClass();
