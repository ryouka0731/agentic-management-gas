/** 種類の見出し。サーバの ANNOUNCEMENT_KINDS と同じ並び */ function ANNOUNCE_KINDS() { return [ { value: 'notice', label: 'お知らせ' }, { value: 'poll', label: 'アンケート' }, { value: 'ad', label: '案内' }, ]; } function announceKindLabel(kind) { var hit = ANNOUNCE_KINDS().filter(function (k) { return k.value === kind; } inner.appendChild(actions); card.appendChild(inner); close.addEventListener('click', function () { announceReply(a, 'closed', buttons); }); homeAnnounceEl.appendChild(card); } /** * 読んだ・答えた・閉じたを送る。押しているあいだはどのボタンも押せなくし、届いたらカードを外す。 * * @param {{id: number}} a * @param {string} answer '' / 'closed' / 選んだ選択肢 * @param {HTMLButtonElement[]} buttons */ function announceReply(a, answer, buttons) { buttons.forEach(function (b) { b.disabled = true; }); call('apiAnnouncementReply', [a.id, answer], function () { var card = homeAnnounceEl.querySelector('[data-announce="' + String(a.id) + '"]'); if (card) homeAnnounceEl.textContent = ''; if (answer !== 'closed') showSnack('ありがとうございました', { kind: 'ok' }); }); loadAnnouncementList(list); return box; } /** 一覧を取って描く。続けて頼んだら、最後に頼んだぶんの返事だけを描く */ function loadAnnouncementList(list) { showSkeleton(list, 'list', 2); var still = latest('announce-list'); call('apiAnnouncementList', [], still.only(function (res) { renderAnnouncementList(list, res); }), still.only(failInto(list))); } /** * 置いたものの一覧 (新しい順)。数は選択肢ごと・読んだ・閉じた。 * * @param {HTMLElement} list * @param {{items: object[]}} res */ function renderAnnouncementList(list, res) { list.textContent = '';
