// ===== 工程表 ===== /** * 期限つきのやることを時系列の棒で並べる。 * * 配置の計算は src/core/Gantt.js の純関数と同じものをここに写している。 * google.script.run の往復を増やさないための意図的な重複であり、 * 片方だけ直すと表示と実体がずれる。 * * @param {object[]} issues * @param {Date} today * @returns {object} */ /** 目盛りの末尾に足す余白の日数。src/core/Gantt.js と同じ値 */ var GANTT_TAIL_DAYS = 14; /** * 予定工数 (人日) を棒の日数に直す。src/core/Gantt.js と同じもの。 * * @param {number|string} planned * @returns {number} */ function ganttDaysForEffort(planned) { var n = Number(planned); if (!n || isNaN(n) || n <= 0) return 1; return Math.max(1, Math.round(n)); } /** * 予定工数と棒の長さが食い違っているかを見る。 * * @param {number|string} planned * @param {number} days * @returns {boolean} */ function ganttEffortDiffers(planned, days) { return ganttDaysForEffort(planned) !== Math.max(1, Number(days) || 1); } function ganttLayout(issues, today) { var DAY = 24 * 60 * 60 * 1000; function startOfDay(value) { if (value === null || value === undefined || value === '') return null; // src/core/Gantt.js の ganttStartOfDay と同じ。'YYYY-MM-DD' はその土地の日 var ymd = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(value)); if (ymd) return new Date(Number(ymd[1]), Number(ymd[2]) - 1, Number(ymd[3])).getTime(); var d = new Date(value); if (isNaN(d.getTime())) return null; return new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime(); } var now = startOfDay(today || new Date()); var rows = []; var from = null; var to = null; (issues || []).forEach(function (issue) { // 開始日を入れていればそれを使う。無ければ作った日を仮の起点にする var planned = startOfDay(issue.startDate); var start = planned !== null ? planned : startOfDay(issue.createdAt); var due = startOfDay(issue.dueDate); if (start === null) return; var end = (due === null) ? start : Math.max(due, start); if (from === null || start < from) from = start; if (to === null || end > to) to = end; rows.push({ number: Number(issue.number), title: issue.title, state: issue.state, assignee: issue.assignee, hasDue: due !== null, hasStart: planned !== null, planned: issue.plannedHours, parent: issue.parent, overdue: due !== null && String(issue.state) === 'open' && due < now, _start: start, _end: end, }); }); if (!rows.length) return { from: now, to: now, days: 1, todayOffset: 0, rows: [] }; if (now < from) from = now; if (now > to) to = now; // 末尾に余白の日を足す。目盛りが最後の期限で終わっていると、棒を // 掴んで伸ばそうとしても伸ばす先の列が無く、縮められるだけになる to += GANTT_TAIL_DAYS * DAY; rows.forEach(function (r) { r.offset = Math.round((r._start - from) / DAY); r.span = Math.round((r._end - r._start) / DAY) + 1; delete r._start; delete r._end; }); return { from: from, to: to, days: Math.round((to - from) / DAY) + 1, todayOffset: Math.round((now - from) / DAY), rows: rows, }; } /** * 月ごとの幅を返す。src/core/Gantt.js と同じもの。 * * @param {number} from * @param {number} days * @returns {Array<{label:string, span:number}>} */ function ganttMonths(from, days) { var DAY = 24 * 60 * 60 * 1000; var out = []; for (var i = 0; i < days; i++) { var d = new Date(from + i * DAY); var year = d.getFullYear(); var month = d.getMonth() + 1; var last = out.length ? out[out.length - 1] : null; if (last && last.year === year && last.month === month) { last.span++; continue; } out.push({ year: year, month: month, span: 1, label: year + '年' + month + '月' }); } return out; } /** * 列の位置を日付に直す。src/core/Gantt.js と同じもの。 * * @param {number} from * @param {number} offset * @returns {string} YYYY-MM-DD */ function ganttDateAt(from, offset) { var d = new Date(from + offset * 24 * 60 * 60 * 1000); var m = String(d.getMonth() + 1); var day = String(d.getDate()); return d.getFullYear() + '-' + (m.length < 2 ? '0' + m : m) + '-' + (day.length < 2 ? '0' + day : day); } /** * やることを束ねる。src/core/Grouping.js と同じもの。 * * @param {object[]} issues * @param {string} by * @returns {Array<{key:string, label:string, issues:object[]}>} */ function groupIssues(issues, by) { var names = {}; allFiles.forEach(function (f) { if (splitPath(f.path).branch === 'main') names[f.fileId] = f.path; }); var order = []; var buckets = Object.create(null); // '__proto__' のような名前も、ただの鍵として持つ function put(key, label, issue) { // 'constructor' のような名前が Object の持ち物と重なる (src/core/Grouping.js と同じ) if (!Object.prototype.hasOwnProperty.call(buckets, key)) { buckets[key] = { key: key, label: label, issues: [] }; order.push(key); } buckets[key].issues.push(issue); } (issues || []).forEach(function (issue) { if (by === 'none') { put('all', 'すべて', issue); return; } if (by === 'assignee') { // 2人で持つ仕事は、両方の束に出す。片方にしか出さないと、 // もう片方の人には自分の仕事が見えなくなる var who = issue.assignees || String(issue.assignee || '').split(',') .map(function (x) { return x.replace(/^\s+|\s+$/g, ''); }) .filter(function (x) { return !!x; }); if (!who.length) { put('(未割当)', '担当なし', issue); return; } who.forEach(function (one) { put(one, personName(one), issue); }); return; } if (by === 'state') { var open = String(issue.state) === 'open'; put(open ?
