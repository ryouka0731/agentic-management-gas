> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ヘッダーに横断検索の欄を置き、文書・やること・確認依頼・改訂中の版・報告・障害物を題名と番号で探して開けるようにする。

**Architecture:** サーバの `src/core/Search.gs` の `searchIndex_()` が全種類の題名・番号・状態を1回の読みで返し、`apiSearchIndex()` が画面へ渡す。画面は欄へ初めて移ったとき (または60秒たったとき) に1回取り、打つたびに手元の純粋関数 (`searchNormalize` / `searchFilter`) で絞って候補を出す。

**Tech Stack:** Google Apps Script (V8)、vitest + `test/fakegas.js`、jsdom (`test/dom.js`)、`test/ui-source.js` (画面の純粋関数を切り出して試す)。

**Spec:** `docs/superpowers/specs/2026-10-14-search-design.md`

## Global Constraints

- ランタイムコードに `import` / `export` / `require` を書かない
- 画面へ返す値は `plainText` / `plainNumber` / `plainId` だけを通す。Date を返さない
- **人の名前・アドレス (author / assignee / by / reportedBy / createdBy) を検索の一覧に入れない**
- 台帳は表ごとに1回だけ読む
- UI は `textContent` のみ (`innerHTML` を使わない)。サーバを呼ぶのは `call()` だけ。CSS は既存トークンのみ
- **返事で画面を切り替えない** (今日2回踏んだ不具合の型)
- 種類ごとに5件まで・全部で20件まで。一覧を取り直す間隔は60秒
- `/` は今までどおりやることの絞り込み欄。Ctrl+K / ⌘K で検索欄へ
- 新しい `.gs` を足したら、`src/Main.gs` を読むテストの SOURCES すべてに足す: `test/e2e-journey.test.js` / `test/phase3-issues.test.js` / `test/phase4-edit.test.js` / `test/phase4-queue.test.js` / `test/priority.test.js` / `test/prclose.test.js` / `test/scrum-api.test.js` (`'src/core/Home.gs'` の直後に `'src/core/Search.gs'`)

```js
/**
 * 横断検索の一覧。設計: docs/superpowers/specs/2026-10-14-search-design.md
 *
 * **題名・番号・状態だけを返す。** 人の名前とアドレスは入れない (一覧で既に見えている題名だけを
 * 探せるようにする。人を探す道は作らない)。画面はこれを1回取り、手元で絞る。
 */

/**
 * @returns {Array<{kind: string, number: number, key: string, title: string, state: string, archived: boolean}>}
 */
function searchIndex_() {
  var out = [];
  function push(kind, number, key, title, state, archived) {
    var t = String(title || '');
    if (!t) return;
    out.push({ kind: kind, number: Number(number) || 0, key: String(key || ''), title: t,
      state: String(state || ''), archived: !!archived });
  }

  var files = filesVisibleInWiki();
  for (var f = 0; f < files.length; f++) {
    var path = String(files[f].path || '');
    if (branchOfPath_(path) !== 'main') continue;
    push('doc', 0, files[f].fileId, path, '', false);
  }

  var issues = dbReadAll('issues');
  for (var i = 0; i < issues.length; i++) {
    push('issue', issues[i].number, '', issues[i].title, issues[i].state, !!issues[i].archivedAt);
  }

  var pulls = dbReadAll('pulls');
  for (var p = 0; p < pulls.length; p++) {
    push('pull', pulls[p].number, '', pulls[p].title, pulls[p].state, false);
  }

  var branches = dbReadAll('branches');
  for (var b = 0; b < branches.length; b++) {
    var name = String(branches[b].name || '');
    if (name === 'main' || String(branches[b].state) !== 'open') continue;
    push('branch', 0, name, name, 'open', false);
  }

  var inquiries = dbReadAll('inquiries');
  for (var q = 0; q < inquiries.length; q++) {
    var title = String(inquiries[q].title || '') || inquiryTitleOf_(inquiries[q].body);
    push('inquiry', inquiries[q].number, '', title, inquiries[q].state, false);
  }

  if (scrumEnabled()) {
    var imps = dbReadAll('impediments');
    for (var m = 0; m < imps.length; m++) {
      push('impediment', imps[m].number, '', imps[m].title, imps[m].state, false);
    }
  }
  return out;
}
```

```js
  // ===== 横断検索 =====

  /** 候補に出す種類の並び */
  var SEARCH_KINDS = ['doc', 'issue', 'pull', 'branch', 'inquiry', 'impediment'];

  /**
   * 比べるための形にそろえる。全角半角・大文字小文字・カタカナとひらがなの違いを消す。
   * **1字ずつ置き換えるので、字の数は変わらない (NFKC で変わる字を除く)。**
   *
   * @param {string} s
   * @returns {string}
   */
  function searchNormalize(s) {
    return String(s || '').normalize('NFKC').toLowerCase()
      .replace(/[ァ-ヶ]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0x60); });
  }

  /**
   * 1件の段。0 番号がぴったり / 1 先頭が合う / 2 途中が合う / -1 合わない。
   *
   * @param {object} item
   * @param {string[]} words 正規化済みの語
   * @returns {number}
   */
  function searchRank(item, words) {
    var title = searchNormalize(item.title);
    var num = words.length === 1 ? words[0].replace(/^#/, '') : '';
    if (num && /^\d+$/.test(num) && Number(num) === Number(item.number) && item.number) return 0;
    for (var i = 0; i < words.length; i++) {
      if (title.indexOf(words[i]) < 0) return -1;
    }
    return title.indexOf(words[0]) === 0 ? 1 : 2;
  }

  /**
   * 候補を種類ごとに並べる。種類ごとに5件・全体で20件まで。
   *
   * @param {object[]} index apiSearchIndex
   * @param {string} query
   * @returns {Array<{kind: string, items: object[]}>}
   */
  function searchFilter(index, query) {
    var words = searchNormalize(query).split(/\s+/).filter(Boolean);
    if (!words.length) return [];
    var stateOrder = { open: 0, approved: 0, closed: 1, merged: 1, done: 1, resolved: 1 };

    var groups = {};
    (index || []).forEach(function (item) {
      var rank = searchRank(item, words);
      if (rank < 0) return;
      // 合った場所 (題名の字の位置)。正規化で字数が変わる字があれば印を付けない
      var hit = null;
      var norm = searchNormalize(item.title);
      if (norm.length === String(item.title).length) {
        var at = norm.indexOf(words[0]);
        if (at >= 0) hit = [at, at + words[0].length];
      }
      (groups[item.kind] = groups[item.kind] || []).push(Object.assign({}, item, { rank: rank, hit: hit }));
    });

    var out = [];
    var left = 20;
    SEARCH_KINDS.forEach(function (kind) {
      var list = groups[kind];
      if (!list || left <= 0) return;
      list.sort(function (a, b) {
        return a.rank - b.rank ||
          (a.archived ? 1 : 0) - (b.archived ? 1 : 0) ||
          (stateOrder[a.state] || 0) - (stateOrder[b.state] || 0) ||
          b.number - a.number;
      });
      var take = list.slice(0, Math.min(5, left));
      left -= take.length;
      out.push({ kind: kind, items: take });
    });
    return out;
  }
```

```js
  var searchBox = document.getElementById('search-box');
  var searchInput = document.getElementById('search-input');
  var searchResults = document.getElementById('search-results');
  var searchIndexRows = null;
  var searchIndexAt = 0;
  var searchActive = -1;
  var searchSeq = 0;
  var SEARCH_LABELS = { doc: '文書', issue: 'やること', pull: '確認依頼', branch: '改訂中の版', inquiry: '報告', impediment: '障害物' };
  var SEARCH_STATES = {
    issue: { open: '作業中', closed: '完了' },
    pull: { open: '確認中', approved: '承認済み', merged: '反映済み', closed: '取り下げ' },
    inquiry: { open: '受付', done: '済み' },
    impediment: { open: '未解決', resolved: '解決' },
  };

  // Mac では ⌘K、ほかは Ctrl+K と案内する
  var searchMac = /Mac|iPhone|iPad/.test(window.navigator.platform || '');
  searchInput.placeholder = '探す (' + (searchMac ? '⌘K' : 'Ctrl+K') + ')';
  document.getElementById('search-open').appendChild(makeIcon('search'));

  /**
   * 一覧を取る。60秒以内に取ったものがあれば取り直さない。
   * **返事で画面を切り替えない。** 届いたとき欄が開いていれば今の字で絞り直す。
   */
  function loadSearchIndex() {
    if (searchIndexRows && Date.now() - searchIndexAt < 60000) return;
    var seq = ++searchSeq;
    call('apiSearchIndex', [], function (rows) {
      if (seq !== searchSeq) return;
      searchIndexRows = rows || [];
      searchIndexAt = Date.now();
      if (document.activeElement === searchInput || searchBox.classList.contains('is-open')) renderSearch();
    }, function (err) {
      if (seq !== searchSeq) return;
      if (document.activeElement !== searchInput && !searchBox.classList.contains('is-open')) return;
      showSearchMessage('探せませんでした: ' + ((err && err.message) || err));
    });
  }

  function closeSearch() {
    searchResults.hidden = true;
    searchResults.textContent = '';
    searchInput.setAttribute('aria-expanded', 'false');
    searchInput.removeAttribute('aria-activedescendant');
    searchActive = -1;
    searchBox.classList.remove('is-open');
  }

  function showSearchMessage(text) {
    searchResults.textContent = '';
    var p = document.createElement('p');
    p.className = 'search-empty row-meta';
    p.textContent = text;
    searchResults.appendChild(p);
    searchResults.hidden = false;
    searchInput.setAttribute('aria-expanded', 'true');
  }

  /** 題名を字のまま並べ、合った部分だけ mark で囲む */
  function searchTitle(item) {
    var span = document.createElement('span');
    span.className = 'search-title';
    var t = String(item.title);
    if (!item.hit) { span.textContent = t; return span; }
    span.appendChild(document.createTextNode(t.substring(0, item.hit[0])));
    var mark = document.createElement('mark');
    mark.textContent = t.substring(item.hit[0], item.hit[1]);
    span.appendChild(mark);
    span.appendChild(document.createTextNode(t.substring(item.hit[1])));
    return span;
  }

  function renderSearch() {
    var q = searchInput.value;
    if (!q.replace(/\s+/g, '')) { closeSearch(); return; }
    if (!searchIndexRows) { showSearchMessage('探しています…'); return; }
    var groups = searchFilter(searchIndexRows, q);
    if (!groups.length) { showSearchMessage('見つかりませんでした'); return; }

    searchResults.textContent = '';
    searchActive = -1;
    var n = 0;
    groups.forEach(function (g) {
      var head = document.createElement('div');
      head.className = 'search-group-head row-meta';
      head.setAttribute('role', 'presentation');
      head.textContent = SEARCH_LABELS[g.kind];
      searchResults.appendChild(head);
      g.items.forEach(function (item) {
        var row = document.createElement('div');
        row.className = 'search-row';
        row.id = 'search-opt-' + (n++);
        row.setAttribute('role', 'option');
        row.setAttribute('aria-selected', 'false');
        row.dataset.kind = item.kind;
        if (item.number) {
          var num = document.createElement('span');
          num.className = 'search-num row-meta';
          num.textContent = '#' + item.number;
          row.appendChild(num);
        }
        row.appendChild(searchTitle(item));
        var label = item.archived ? '置き場' : ((SEARCH_STATES[item.kind] || {})[item.state] || '');
        if (label) row.appendChild(makePill(label));
        row.addEventListener('mousedown', function (e) { e.preventDefault(); });
        row.addEventListener('click', function () { openSearchItem(item); });
        searchResults.appendChild(row);
      });
    });
    searchResults.hidden = false;
    searchInput.setAttribute('aria-expanded', 'true');
  }

  function moveSearch(step) {
    var rows = searchResults.querySelectorAll('.search-row');
    if (!rows.length) return;
    searchActive = Math.max(0, Math.min(rows.length - 1, searchActive + step));
    for (var i = 0; i < rows.length; i++) rows[i].setAttribute('aria-selected', String(i === searchActive));
    searchInput.setAttribute('aria-activedescendant', rows[searchActive].id);
    if (rows[searchActive].scrollIntoView) rows[searchActive].scrollIntoView({ block: 'nearest' });
  }

  /** 選んだものを開く。開いたら欄を空にして閉じる */
  function openSearchItem(item) {
    searchInput.value = '';
    closeSearch();
    searchInput.blur();
    if (item.kind === 'doc') openDocById(item.key);
    else if (item.kind === 'issue') openIssueByNumber(item.number);
    else if (item.kind === 'pull') switchTab('pulls', { pr: item.number });
    else if (item.kind === 'branch') switchTab('branches');
    else if (item.kind === 'inquiry') { reportShown = item.number; switchTab('report'); }
    else if (item.kind === 'impediment') switchTab('impediments');
  }

  searchInput.addEventListener('focus', loadSearchIndex);
  searchInput.addEventListener('input', renderSearch);
  searchInput.addEventListener('blur', function () {
    // 候補を押したときは mousedown で blur を止めてある
    if (!searchBox.classList.contains('is-open')) closeSearch();
  });
  searchInput.addEventListener('keydown', function (e) {
    if (e.key === 'ArrowDown') { e.preventDefault(); moveSearch(1); }
    else if (e.key === 'ArrowUp') { e.preventDefault(); moveSearch(-1); }
    else if (e.key === 'Enter') {
      var rows = searchResults.querySelectorAll('.search-row');
      var pick = rows[searchActive >= 0 ? searchActive : 0];
      if (pick) { e.preventDefault(); pick.click(); }
    } else if (e.key === 'Escape') {
      e.preventDefault();
      e.stopPropagation();
      closeSearch();
      searchInput.blur();
    }
  });

  document.getElementById('search-open').addEventListener('click', function () {
    searchBox.classList.add('is-open');
    searchInput.focus();
    loadSearchIndex();
  });
  document.getElementById('search-close').addEventListener('click', function () {
    searchInput.value = '';
    closeSearch();
  });

  // Ctrl+K / ⌘K でどこからでも欄へ (入力欄の中でも効く)
  document.addEventListener('keydown', function (e) {
    if ((e.ctrlKey || e.metaKey) && !e.altKey && (e.key === 'k' || e.key === 'K')) {
      e.preventDefault();
      if (window.matchMedia && window.matchMedia('(max-width: 767px)').matches) searchBox.classList.add('is-open');
      searchInput.focus();
      loadSearchIndex();
    }
  });
```
